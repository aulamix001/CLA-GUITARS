cmake_minimum_required(VERSION 3.22)
project(CLAInspiredGuitar VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(BUILD_PLUGIN "Build JUCE VST3 and standalone" ON)
include(CTest)
add_executable(dsp_tests tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME dsp_tests COMMAND dsp_tests)
if(BUILD_PLUGIN)
  if(DEFINED JUCE_DIR)
    find_package(JUCE CONFIG REQUIRED)
  elseif(DEFINED JUCE_PATH)
    add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  if(CMAKE_CROSSCOMPILING)
    set(rack_manifest FALSE)
  else()
    set(rack_manifest TRUE)
  endif()
  juce_add_plugin(CLAInspiredGuitar COMPANY_NAME "Independent Audio" BUNDLE_ID "audio.independent.guitarrack" PLUGIN_MANUFACTURER_CODE InAu PLUGIN_CODE Clg1 FORMATS VST3 Standalone PRODUCT_NAME "Guitar Rack" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST ${rack_manifest})
  target_sources(CLAInspiredGuitar PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
  target_compile_definitions(CLAInspiredGuitar PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(CLAInspiredGuitar PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endif()

option(BUILD_UI_PREVIEW "Build the interface render and interaction check" OFF)
if(BUILD_UI_PREVIEW AND BUILD_PLUGIN)
  juce_add_console_app(RackPreview PRODUCT_NAME "RackPreview")
  target_sources(RackPreview PRIVATE tests/Preview.cpp)
  target_include_directories(RackPreview PRIVATE Source)
  target_link_libraries(RackPreview PRIVATE CLAInspiredGuitar juce::juce_audio_utils)
  target_compile_definitions(RackPreview PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
